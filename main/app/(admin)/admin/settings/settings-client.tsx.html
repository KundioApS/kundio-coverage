function Toggle({
  on,
  onChange,
  disabled = false,
}: {
  on: boolean;
  onChange: (val: boolean) => void;
  disabled?: boolean;
}) {
  return (
    <button
      type="button"
      onClick={() => {
        if (!disabled) onChange(!on);
      }}
      disabled={disabled}
      style={{
        appearance: "none",
        border: 0,
        padding: 0,
        cursor: disabled ? "not-allowed" : "pointer",
        width: 38,
        height: 22,
        borderRadius: 999,
        background: on ? "var(--brand)" : "var(--line-strong)",
        position: "relative",
        transition: "background 150ms",
        flexShrink: 0,
        opacity: disabled ? 0.6 : 1,
      }}
    >
      <span
        style={{
          position: "absolute",
          top: 2,
          left: on ? 18 : 2,
          width: 18,
          height: 18,
          borderRadius: "50%",
          background: "white",
          transition: "left 150ms",
          boxShadow: "0 1px 3px rgba(0,0,0,.2)",
        }}
      />
    </button>
  );
}

export function SettingsView({
  initialSettings,

  const [notice, setNotice] = useState<{ tone: "ok" | "danger" | "info"; text: string } | null>(null);

  async function save() {
    setSaving(true);
    setNotice(null);
    const payload = {
      organizationName: settings.organizationName,
      supportEmail: settings.supportEmail,
      selfBusinessId: settings.selfBusinessId,
      timezone: settings.timezone,
      defaultLanguage: settings.defaultLanguage,
      defaultSeatLimit: settings.defaultSeatLimit,
      defaultMonthlyLeadLimit: settings.defaultMonthlyLeadLimit,
      defaultMonthlyMessageLimit: settings.defaultMonthlyMessageLimit,
      autoSendInviteEmail: settings.autoSendInviteEmail,
      twilioProvisioningMode: settings.twilioProvisioningMode,
      twilioAutoProvisionOnActivation: settings.twilioAutoProvisionOnActivation,
      twilioParentAccountSid: settings.twilioParentAccountSid,
      stripeDefaultCurrency: settings.stripeDefaultCurrency,
      stripeVatRateBps: settings.stripeVatRateBps,
      stripeInvoiceFooter: settings.stripeInvoiceFooter,
      confirmDestructiveActions: settings.confirmDestructiveActions,
      notifyFailedPayment: settings.notifyFailedPayment,
      notifyTwilioFailure: settings.notifyTwilioFailure,
      notifyNewSignup: settings.notifyNewSignup,
      ...(settings.slackWebhookUrl.trim()
        ? { slackWebhookUrl: settings.slackWebhookUrl.trim() }
        : savedSettings.slackWebhookConfigured
          ? {}
          : { slackWebhookUrl: null }),
    };
 
    try {
      const res = await fetch("/api/admin/settings", {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      });
      const json = await res.json();
      if (!res.ok) throw new Error(json.error ?? "Kunne ikke gemme indstillinger");
      setSettings(json.data);
      setSavedSettings(json.data);
      setNotice({ tone: "ok", text: "Indstillingerne er gemt." });
    } catch (err) {
      setNotice({
        tone: "danger",
        text: err instanceof Error ? err.message : "Kunne ikke gemme indstillinger.",
      });
    } finally {
      setSaving(false);
    }
  }
