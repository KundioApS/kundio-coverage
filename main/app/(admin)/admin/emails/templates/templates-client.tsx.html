"use client";
 
import {
  Archive,
  Copy,
  FileText,
  Inbox,
  Mail,
  Newspaper,
  Search,
  SquarePen,
} from "lucide-react";
import Link from "next/link";
import { useRouter } from "next/navigation";
import { useMemo, useState, useTransition } from "react";
 
export type TemplateRow = {
  id: string;
  templateType: "admin_email" | "newsletter";
  name: string;
  slug: string;
  description: string | null;
  subject: string;
  category: string | null;
  isActive: boolean;
  archivedAt: string | null;
  updatedAt: string;
};
 
type Props = {
  initial: TemplateRow[];
};
 
type TypeFilter = "all" | TemplateRow["templateType"];
type StatusFilter = "all" | "active" | "inactive" | "archived";
 
const typeLabels: Record<TypeFilter, string> = {
  all: "Alle",
  admin_email: "Kundeemail",
  newsletter: "Nyhedsbrev",
};
 
const statusLabels: Record<StatusFilter, string> = {
  all: "Alle",
  active: "Aktive",
  inactive: "Inaktive",
  archived: "Arkiverede",
};
 
function formatDateTime(value: string | null | undefined) {
  if (!value) return "Ikke endnu";
  return new Intl.DateTimeFormat("da-DK", {
    dateStyle: "medium",
    timeStyle: "short",
  }).format(new Date(value));
}
 
function typeClass(type: TemplateRow["templateType"]) {
  return type === "newsletter" ? "pill pill-pending" : "pill pill-neutral";
}
 
function statusClass(row: TemplateRow) {
  if (row.archivedAt) return "pill pill-neutral";
  return row.isActive ? "pill pill-live" : "pill pill-pending";
}
 
function statusLabel(row: TemplateRow) {
  if (row.archivedAt) return "arkiveret";
  return row.isActive ? "aktiv" : "inaktiv";
}
 
function matchesStatus(row: TemplateRow, status: StatusFilter) {
  if (status === "all") return true;
  if (status === "archived") return Boolean(row.archivedAt);
  if (status === "active") return row.isActive && !row.archivedAt;
  return !row.isActive && !row.archivedAt;
}
 
export function TemplatesList({ initial }: Props) {
  const router = useRouter();
  const [rows, setRows] = useState(initial);
  const [query, setQuery] = useState("");
  const [typeFilter, setTypeFilter] = useState<TypeFilter>("all");
  const [statusFilter, setStatusFilter] = useState<StatusFilter>("all");
  const [pendingId, setPendingId] = useState<string | null>(null);
  const [pending, startTransition] = useTransition();
  const [error, setError] = useState<string | null>(null);
 
  async function onDuplicate(id: string) {
    setError(null);
    setPendingId(id);
    try {
      const res = await fetch(`/api/admin/emails/templates/${id}/duplicate`, {
        method: "POST",
      });
      const json = await res.json().catch(() => ({}));
      if (!res.ok) {
        setError(json.error ?? "Kunne ikke duplikere.");
        return;
      }
      startTransition(() => router.refresh());
    } finally {
      setPendingId(null);
    }
  }
 
  async function onArchive(id: string) {
    if (!confirm("Arkivér denne skabelon?")) return;
    setError(null);
    setPendingId(id);
    try {
      const res = await fetch(`/api/admin/emails/templates/${id}`, {
        method: "DELETE",
      });
      const json = await res.json().catch(() => ({}));
      if (!res.ok) {
        setError(json.error ?? "Kunne ikke arkivere.");
        return;
      }
      setRows((prev) =>
        prev.map((row) =>
          row.id === id
            ? {
                ...row,
                isActive: false,
                archivedAt: new Date().toISOString(),
              }
            : row,
        ),
      );
    } finally {
      setPendingId(null);
    }
  }
 
  const filtered = useMemo(() => {
    const needle = query.trim().toLowerCase();
    return rows.filter((row) => {
      const matchesType = typeFilter === "all" || row.templateType === typeFilter;
      const matchesState = matchesStatus(row, statusFilter);
      const matchesQuery =
        !needle ||
        row.name.toLowerCase().includes(needle) ||
        row.slug.toLowerCase().includes(needle) ||
        row.subject.toLowerCase().includes(needle) ||
        row.category?.toLowerCase().includes(needle) ||
        row.description?.toLowerCase().includes(needle);
 
      return matchesType && matchesState && matchesQuery;
    });
  }, [query, rows, statusFilter, typeFilter]);
 
  return (
    <section className="email-panel">
      <div className="email-panel-header">
        <div className="email-panel-title">
          <h2>Skabelonbibliotek</h2>
          <p>Søg i genbrugeligt indhold, se status, og åbn editoren.</p>
        </div>
        <Link href="/admin/emails/templates/new" className="btn btn-brand btn-sm">
          <FileText />
          Ny skabelon
        </Link>
      </div>
 
      <div className="email-panel-body">
        <div className="email-list-toolbar">
          <div className="email-filter-group">
            <div className="email-filter-search">
              <Search />
              <input
                className="input"
                name="template-search"
                autoComplete="off"
                value={query}
                onChange={(event) => setQuery(event.target.value)}
                placeholder="Søg navn, emne eller slug…"
              />
            </div>
            <div className="seg" role="group" aria-label="Filter for skabelontype">
              {(Object.keys(typeLabels) as TypeFilter[]).map((key) => (
                <button
                  key={key}
                  type="button"
                  className={typeFilter === key ? "on" : ""}
                  onClick={() => setTypeFilter(key)}
                >
                  {typeLabels[key]}
                  <span className="seg-count">
                    {key === "all"
                      ? rows.length
                      : rows.filter((row) => row.templateType === key).length}
                  </span>
                </button>
              ))}
            </div>
            <div className="seg" role="group" aria-label="Filter for skabelonstatus">
              {(Object.keys(statusLabels) as StatusFilter[]).map((key) => (
                <button
                  key={key}
                  type="button"
                  className={statusFilter === key ? "on" : ""}
                  onClick={() => setStatusFilter(key)}
                >
                  {statusLabels[key]}
                  <span className="seg-count">
                    {key === "all"
                      ? rows.length
                      : rows.filter((row) => matchesStatus(row, key)).length}
                  </span>
                </button>
              ))}
            </div>
          </div>
        </div>
 
        {error ? <div className="notice notice-danger">{error}</div> : null}
 
        {rows.length === 0 ? (
          <div className="email-empty-panel">
            <span className="email-empty-icon">
              <Inbox />
            </span>
            <h3>Ingen skabeloner endnu</h3>
            <p>
              Opret den første genbrugelige skabelon til kundeemails eller
              nyhedsbrevsudkast.
            </p>
            <Link href="/admin/emails/templates/new" className="btn btn-brand">
              Ny skabelon
            </Link>
          </div>
        ) : filtered.length === 0 ? (
          <div className="email-empty-panel">
            <span className="email-empty-icon">
              <Search />
            </span>
            <h3>Ingen matchende skabeloner</h3>
            <p>Juster søgeteksten, typefilteret eller statusfilteret.</p>
          </div>
        ) : (
          <div className="email-card-list">
            {filtered.map((row) => (
              <article key={row.id} className="email-template-row">
                <div className="email-row-main">
                  <div className="email-row-title">
                    <Link href={`/admin/emails/templates/${row.id}`}>
                      <b>{row.name}</b>
                    </Link>
                    <span className={typeClass(row.templateType)}>
                      <span className="pill-dot" />
                      {row.templateType === "newsletter" ? "nyhedsbrev" : "kundeemail"}
                    </span>
                    <span className={statusClass(row)}>
                      <span className="pill-dot" />
                      {statusLabel(row)}
                    </span>
                  </div>
                  <div className="email-row-note">{row.subject}</div>
                  <div className="email-row-meta">
                    <span>{row.slug}</span>
                    <span>{row.category ?? "Ingen kategori"}</span>
                    <span>Opdateret {formatDateTime(row.updatedAt)}</span>
                  </div>
                  {row.description ? (
                    <div className="email-row-description">{row.description}</div>
                  ) : null}
                </div>
                <div className="email-status-stack">
                  <div className="email-compact-stats">
                    <div className="email-compact-stat">
                      <span>Type</span>
                      <b>
                        {row.templateType === "newsletter" ? (
                          <Newspaper aria-hidden="true" />
                        ) : (
                          <Mail aria-hidden="true" />
                        )}
                      </b>
                    </div>
                    <div className="email-compact-stat">
                      <span>Status</span>
                      <b>{statusLabel(row)}</b>
                    </div>
                  </div>
                  <div className="email-row-actions">
                    <Link
                      href={`/admin/emails/templates/${row.id}`}
                      className="btn btn-outline btn-sm"
                    >
                      <SquarePen />
                      Rediger
                    </Link>
                    <button
                      className="btn btn-outline btn-sm"
                      type="button"
                      onClick={() => onDuplicate(row.id)}
                      disabled={pending || pendingId === row.id}
                    >
                      <Copy />
                      Duplikér
                    </button>
                    {!row.archivedAt ? (
                      <button
                        className="btn btn-danger-outline btn-sm"
                        type="button"
                        onClick={() => onArchive(row.id)}
                        disabled={pending || pendingId === row.id}
                      >
                        <Archive />
                        Arkivér
                      </button>
                    ) : null}
                  </div>
                </div>
              </article>
            ))}
          </div>
        )}
      </div>
    </section>
  );
}
