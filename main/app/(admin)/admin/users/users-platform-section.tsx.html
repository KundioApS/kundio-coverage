"use client";
 
import { useEffect, useMemo, useRef, useState } from "react";
import { Building2, Users } from "lucide-react";
import Link from "next/link";
import type {
  AdminPageInfo,
  PlatformInvite,
  PlatformUser,
} from "@/lib/admin/users";
import { fmtDate, fmtDateTime } from "../_lib/format";
import { useSort, sortRows, SortableHeader } from "../_components/sort-header";
import { TablePagination, type PageSize } from "../_components/table-pagination";
import {
  includesQuery,
  initials,
  inviteStatusPill,
  platformStatusPill,
} from "./users-view-shared";
 
export type PlatformUsersSectionProps = {
  initialUsers: PlatformUser[];
  initialInvites: PlatformInvite[];
  initialUsersPage: AdminPageInfo;
  initialInvitesPage: AdminPageInfo;
  query: string;
};
 
export function PlatformUsersSection({
  initialUsers,
  initialInvites,
  initialUsersPage,
  initialInvitesPage,
  query,
}: PlatformUsersSectionProps) {
  const { sort: userSort, toggle: toggleUser } = useSort();
  const [users, setUsers] = useState(initialUsers);
  const [invites, setInvites] = useState(initialInvites);
  const [usersPage, setUsersPage] = useState(initialUsersPage);
  const [invitesPage, setInvitesPage] = useState(initialInvitesPage);
  const [userPage, setUserPage] = useState(initialUsersPage.page);
  const [userPerPage, setUserPerPage] = useState<PageSize>(
    initialUsersPage.perPage as PageSize,
  );
  const [invitePage, setInvitePage] = useState(initialInvitesPage.page);
  const [invitePerPage, setInvitePerPage] = useState<PageSize>(
    initialInvitesPage.perPage as PageSize,
  );
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const loadedRequestKeyRef = useRef(
    new URLSearchParams({
      page: String(initialUsersPage.page),
      per_page: String(initialUsersPage.perPage),
      invites_page: String(initialInvitesPage.page),
      invites_per_page: String(initialInvitesPage.perPage),
    }).toString(),
  );
 
  useEffect(() => {
    const controller = new AbortController();
    const params = new URLSearchParams({
      page: String(userPage),
      per_page: String(userPerPage),
      invites_page: String(invitePage),
      invites_per_page: String(invitePerPage),
    });
    const requestKey = params.toString();
    if (loadedRequestKeyRef.current === requestKey) {
      return;
    }
 
    queueMicrotask(() => setIsLoading(true));
    fetch(`/api/admin/platform-users?${params.toString()}`, {
      signal: controller.signal,
    })
      .then(async (res) => {
        const json = await res.json();
        if (!res.ok) throw new Error(json.error ?? "Kunne ikke hente platformbrugere");
        setUsers(json.data.users);
        setInvites(json.data.pending_invites);
        setUsersPage(json.data.users_page);
        setInvitesPage(json.data.invites_page);
        loadedRequestKeyRef.current = requestKey;
        setError(null);
      })
      .catch((err) => {
        Eif (err.name !== "AbortError") {
          setError(err instanceof Error ? err.message : "Kunne ikke hente platformbrugere");
        }
      })
      .finally(() => setIsLoading(false));
 
    return () => controller.abort();
  }, [invitePage, invitePerPage, userPage, userPerPage]);
 
  const filteredUsers = useMemo(
    () =>
      users.filter((user) =>
        includesQuery(
          [
            user.email,
            user.id,
            ...user.memberships.flatMap((membership) => [
              membership.businessName,
              membership.businessId,
              membership.role,
            ]),
          ],
          query,
        ),
      ),
    [query, users],
  );
 
  const filteredInvites = useMemo(
    () =>
      invites.filter((invite) =>
        includesQuery(
          [invite.email, invite.businessName, invite.businessId, invite.role],
          query,
        ),
      ),
    [invites, query],
  );
 
  const sortedUsers = useMemo(
    () =>
      sortRows(filteredUsers, userSort, (u, key) => {
        switch (key) {
          case "email": return u.email;
          case "status": return u.memberships.length > 0 ? "member" : u.confirmedAt ? "no_company" : "unconfirmed";
          case "lastActive": return u.lastSignInAt ?? "";
          case "created": return u.createdAt;
          default: return null;
        }
      }),
    [filteredUsers, userSort],
  );
 
  const queryActive = query.trim() !== "";
  const usersTotal = queryActive ? sortedUsers.length : usersPage.total;
  const invitesTotal = queryActive ? filteredInvites.length : invitesPage.total;
 
  const shProps = {
    sort: userSort,
    onSort: (k: string) => {
      toggleUser(k);
      setUserPage(1);
    },
  };
 
  return (
    <div className="stack">
      <div className="identity-note identity-note-platform">
        <Users />
        <div>
          <b>Platformidentitet</b>
          <span>
            Disse rækker kommer fra Supabase Auth, <code>business_memberships</code>{" "}
            og <code>business_invites</code>. Tabellen er read-only fra admin v2.
          </span>
        </div>
      </div>
 
      {error && <div className="notice notice-danger">{error}</div>}
 
      <div className="card card-bd-tight">
        {sortedUsers.length === 0 ? (
          <div className="empty-state">
            <h3>Ingen platformbrugere fundet</h3>
            <p>Prøv at justere søgningen.</p>
          </div>
        ) : (
          <div className="tbl-wrap">
            <table className="tbl tbl-name-table">
              <thead>
                <tr>
                  <SortableHeader label="Platformbruger" sortKey="email" {...shProps} />
                  <th>Virksomheder</th>
                  <SortableHeader label="Status" sortKey="status" style={{ minWidth: 120 }} {...shProps} />
                  <SortableHeader label="Sidst aktiv" sortKey="lastActive" style={{ minWidth: 120 }} {...shProps} />
                  <SortableHeader label="Oprettet" sortKey="created" style={{ minWidth: 100 }} {...shProps} />
                </tr>
              </thead>
              <tbody>
                {sortedUsers.map((user) => (
                  <tr key={user.id}>
                    <td>
                      <div className="tbl-name">
                        <div className="tbl-avatar">{initials(user.email)}</div>
                        <div className="tbl-name-text">
                          <b>{user.email}</b>
                          <small>{user.id}</small>
                        </div>
                      </div>
                    </td>
                    <td>
                      <div className="membership-list">
                        {user.memberships.length === 0 ? (
                          <span className="empty">Ingen medlemskaber</span>
                        ) : (
                          user.memberships.map((membership) => (
                            <Link
                              key={membership.id}
                              href={`/admin/${membership.businessId}`}
                              className="membership-chip"
                            >
                              <Building2 />
                              {membership.businessName}
                              <span>{membership.role}</span>
                            </Link>
                          ))
                        )}
                      </div>
                    </td>
                    <td>{platformStatusPill(user)}</td>
                    <td className="tabnum">{fmtDateTime(user.lastSignInAt)}</td>
                    <td className="tabnum">{fmtDate(user.createdAt)}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </div>
 
      <TablePagination
        state={{
          page: queryActive ? 1 : usersPage.page,
          perPage: userPerPage,
          total: usersTotal,
        }}
        onPageChange={setUserPage}
        onPerPageChange={(next) => {
          setUserPerPage(next);
          setUserPage(1);
        }}
      >
        {isLoading && <span className="muted">Henter...</span>}
      </TablePagination>
 
      <div className="card card-bd-tight">
        <div className="card-hd">
          <div className="card-hd-text">
            <h3>Afventende platforminvitationer</h3>
            <p>Read-only visning af virksomhedsmedlemskaber der endnu ikke er accepteret.</p>
          </div>
        </div>
        {filteredInvites.length === 0 ? (
          <div className="empty-state">
            <h3>Ingen afventende platforminvitationer</h3>
            <p>Der er ingen pending company invites i den aktuelle visning.</p>
          </div>
        ) : (
          <div className="tbl-wrap">
            <table className="tbl tbl-name-table">
              <thead>
                <tr>
                  <th>Invitation</th>
                  <th>Virksomhed</th>
                  <th>Rolle</th>
                  <th>Status</th>
                  <th>Udløber</th>
                </tr>
              </thead>
              <tbody>
                {filteredInvites.map((invite) => (
                  <tr key={invite.id}>
                    <td>
                      <div className="tbl-name-text">
                        <b>{invite.email}</b>
                        <small>{invite.id}</small>
                      </div>
                    </td>
                    <td>
                      <Link
                        href={`/admin/${invite.businessId}`}
                        className="membership-chip"
                      >
                        <Building2 />
                        {invite.businessName}
                      </Link>
                    </td>
                    <td>
                      <span className="pill pill-neutral">{invite.role}</span>
                    </td>
                    <td>{inviteStatusPill(invite)}</td>
                    <td className="tabnum">{fmtDate(invite.expiresAt)}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </div>
 
      <TablePagination
        state={{
          page: queryActive ? 1 : invitesPage.page,
          perPage: invitePerPage,
          total: invitesTotal,
        }}
        onPageChange={setInvitePage}
        onPerPageChange={(next) => {
          setInvitePerPage(next);
          setInvitePage(1);
        }}
      />
    </div>
  );
}
