import { redirect } from "next/navigation";
import { createAdminClient } from "@/lib/supabase/admin";
import { getServerContext } from "@/lib/server-context";
import { AnalyticsClient } from "@/components/analytics/analytics-client";
import type { Database } from "@/types/database";
 
type LeadStatus = Database["public"]["Enums"]["lead_status"];
 
export interface LeadTrendPoint      { day: string; leads: number }
export interface MsgTrendPoint       { day: string; inbound: number; outbound: number }
export interface StatusPoint         { status: LeadStatus; label: string; count: number; pct: number }
export interface OriginPoint         { name: string; value: number }
export interface ConvControllerPoint { name: string; value: number }
export interface DeliveryPoint       { status: string; label: string; value: number }
 
export interface AnalyticsData {
  kpis: {
    totalLeads: number;
    leadsThisMonth: number;
    smsSentMonth: number;
    aiReplyCount: number;
    handoffCount: number;
    conversionRate: number;
    openConversations: number;
    optedOutConversations: number;
  };
  leadTrend:               LeadTrendPoint[];
  messageTrend:            MsgTrendPoint[];
  statusBreakdown:         StatusPoint[];
  messageOrigins:          OriginPoint[];
  convControllerBreakdown: ConvControllerPoint[];
  deliveryBreakdown:       DeliveryPoint[];
}
 
const STATUS_LABELS: Record<LeadStatus, string> = {
  new: "Ny", contacted: "Kontaktet", in_progress: "I gang",
  booked: "Booket", closed: "Lukket", lost: "Mistet",
};
 
function startOfMonth() {
  const d = new Date();
  d.setDate(1);
  d.setHours(0, 0, 0, 0);
  return d.toISOString();
}
 
function daysAgo(n: number) {
  return new Date(Date.now() - n * 24 * 60 * 60 * 1000).toISOString();
}
 
function dayBuckets(n: number): Record<string, number> {
  const b: Record<string, number> = {};
  const now = new Date();
  for (let i = n - 1; i >= 0; i--) {
    const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000);
    b[d.toISOString().slice(0, 10)] = 0;
  }
  return b;
}
 
function fmtDay(iso: string) {
  return new Date(iso).toLocaleDateString("da-DK", { day: "numeric", month: "short" });
}
 
export default async function AnalyticsPage() {
  const { user, ctx } = await getServerContext();
  if (!user) redirect("/auth/login");
 
  if (!ctx) {
    return (
      <div className="p-6 text-sm" style={{ color: "var(--muted-foreground)" }}>
        Ingen virksomhedsprofil fundet.
      </div>
    );
  }
 
  const admin = createAdminClient();
  const monthStart = startOfMonth();
  const thirtyDaysAgo = daysAgo(30);
 
  // Fetch conversations once — derive lifecycle/controller counts, reuse IDs for message queries
  const { data: convRows } = await admin
    .from("conversations")
    .select("id, lifecycle_status, controlled_by")
    .eq("business_id", ctx.businessId);
  const convIds      = (convRows ?? []).map((c) => c.id);
  const hasConvs     = convIds.length > 0;
  const openConvCount    = (convRows ?? []).filter((r) => r.lifecycle_status === "open").length;
  const optedOutCount    = (convRows ?? []).filter((r) => r.lifecycle_status === "opted_out").length;
 
  // Lead-only queries can run immediately
  const [
    { count: totalLeads },
    { count: leadsThisMonth },
    { data: leadStatusRows },
    { data: dailyLeadRows },
    { count: handoffCount },
  ] = await Promise.all([
    admin.from("leads").select("id", { count: "exact", head: true }).eq("business_id", ctx.businessId),
    admin.from("leads").select("id", { count: "exact", head: true }).eq("business_id", ctx.businessId).gte("created_at", monthStart),
    admin.from("leads").select("status").eq("business_id", ctx.businessId),
    admin.from("leads").select("created_at").eq("business_id", ctx.businessId).gte("created_at", thirtyDaysAgo),
    admin.from("conversation_events").select("id", { count: "exact", head: true }).eq("business_id", ctx.businessId).eq("event_type", "conversation.handoff_requested").gte("created_at", thirtyDaysAgo),
  ]);
 
  // Message queries only make sense when conversations exist
  let smsSentMonth = 0;
  let aiReplyCount = 0;
  let dailyMsgRows:  Array<{ created_at: string; direction: string }> = [];
  let originRows:    Array<{ origin: string }> = [];
  let deliveryRows:  Array<{ delivery_status: string }> = [];
 
  if (hasConvs) {
    const [smsRes, aiRes, msgRes, origRes, delivRes] = await Promise.all([
      admin.from("messages").select("id", { count: "exact", head: true }).eq("direction", "outbound").gte("created_at", monthStart).in("conversation_id", convIds),
      admin.from("messages").select("id", { count: "exact", head: true }).eq("origin", "ai").eq("direction", "outbound").gte("created_at", thirtyDaysAgo).in("conversation_id", convIds),
      admin.from("messages").select("created_at, direction").gte("created_at", thirtyDaysAgo).in("conversation_id", convIds),
      admin.from("messages").select("origin").eq("direction", "outbound").gte("created_at", thirtyDaysAgo).in("conversation_id", convIds),
      admin.from("messages").select("delivery_status").eq("direction", "outbound").gte("created_at", thirtyDaysAgo).in("conversation_id", convIds),
    ]);
    smsSentMonth  = smsRes.count  ?? 0;
    aiReplyCount  = aiRes.count   ?? 0;
    dailyMsgRows  = (msgRes.data  ?? []) as typeof dailyMsgRows;
    originRows    = (origRes.data ?? []) as typeof originRows;
    deliveryRows  = (delivRes.data ?? []) as typeof deliveryRows;
  }
 
  // --- Build status breakdown ---
  const statusCounts: Partial<Record<LeadStatus, number>> = {};
  for (const row of leadStatusRows ?? []) {
    statusCounts[row.status] = (statusCounts[row.status] ?? 0) + 1;
  }
  const tot = totalLeads ?? 0;
  const statusBreakdown: StatusPoint[] = (Object.keys(STATUS_LABELS) as LeadStatus[]).map((s) => ({
    status: s,
    label:  STATUS_LABELS[s],
    count:  statusCounts[s] ?? 0,
    pct:    tot > 0 ? Math.round(((statusCounts[s] ?? 0) / tot) * 100) : 0,
  }));
 
  // --- Build 30-day lead trend ---
  const leadBuckets = dayBuckets(30);
  for (const row of dailyLeadRows ?? []) {
    const day = row.created_at.slice(0, 10);
    if (day in leadBuckets) leadBuckets[day]++;
  }
  const leadTrend: LeadTrendPoint[] = Object.entries(leadBuckets).map(([date, leads]) => ({
    day: fmtDay(date),
    leads,
  }));
 
  // --- Build 30-day message trend ---
  const inboundBuckets  = dayBuckets(30);
  const outboundBuckets = dayBuckets(30);
  for (const row of dailyMsgRows) {
    const day = row.created_at?.slice(0, 10);
    if (!day) continue;
    if (row.direction === "inbound"  && day in inboundBuckets)  inboundBuckets[day]++;
    if (row.direction === "outbound" && day in outboundBuckets) outboundBuckets[day]++;
  }
  const messageTrend: MsgTrendPoint[] = Object.keys(leadBuckets).map((date) => ({
    day:      fmtDay(date),
    inbound:  inboundBuckets[date]  ?? 0,
    outbound: outboundBuckets[date] ?? 0,
  }));
 
  // --- Build message origins ---
  const originCounts: Record<string, number> = {};
  for (const row of originRows) {
    const o = row.origin ?? "system";
    originCounts[o] = (originCounts[o] ?? 0) + 1;
  }
  const ORIGIN_LABELS: Record<string, string> = { ai: "AI", human: "Menneske", system: "System" };
  const messageOrigins: OriginPoint[] = Object.entries(originCounts)
    .map(([o, v]) => ({ name: ORIGIN_LABELS[o] ?? o, value: v }))
    .filter((d) => d.value > 0);
 
  const conversionRate = tot > 0
    ? Math.round((((statusCounts.booked ?? 0) + (statusCounts.closed ?? 0)) / tot) * 100)
    : 0;
 
  // --- Build conversation controller breakdown (open convs only) ---
  const controllerCounts: Record<string, number> = {};
  for (const row of convRows ?? []) {
    if (row.lifecycle_status !== "open") continue;
    const c = row.controlled_by ?? "system";
    controllerCounts[c] = (controllerCounts[c] ?? 0) + 1;
  }
  const CONTROLLER_LABELS: Record<string, string> = { ai: "AI", human: "Menneske", system: "System" };
  const convControllerBreakdown: ConvControllerPoint[] = Object.entries(controllerCounts)
    .map(([c, v]) => ({ name: CONTROLLER_LABELS[c] ?? c, value: v }))
    .filter((d) => d.value > 0);
 
  // --- Build delivery status breakdown (outbound, 30d) ---
  const DELIVERY_LABELS: Record<string, string> = {
    delivered: "Leveret", sent: "Sendt", failed: "Fejlet", queued: "Kø",
  };
  const deliveryCounts: Record<string, number> = {};
  for (const row of deliveryRows) {
    const s = row.delivery_status ?? "unknown";
    deliveryCounts[s] = (deliveryCounts[s] ?? 0) + 1;
  }
  const deliveryBreakdown: DeliveryPoint[] = Object.entries(deliveryCounts)
    .map(([s, v]) => ({ status: s, label: DELIVERY_LABELS[s] ?? s, value: v }))
    .filter((d) => d.value > 0);
 
  const data: AnalyticsData = {
    kpis: {
      totalLeads: tot,
      leadsThisMonth: leadsThisMonth ?? 0,
      smsSentMonth,
      aiReplyCount,
      handoffCount: handoffCount ?? 0,
      conversionRate,
      openConversations: openConvCount,
      optedOutConversations: optedOutCount,
    },
    leadTrend,
    messageTrend,
    statusBreakdown,
    messageOrigins,
    convControllerBreakdown,
    deliveryBreakdown,
  };
 
  return (
    <div className="min-h-full">
      <div className="max-w-[1240px] mx-auto px-7 pt-7 pb-24">
        <div style={{ marginBottom: "22px" }}>
          <h1
            className="font-semibold leading-[1.15] tracking-[-0.02em]"
            style={{ fontSize: "26px", margin: "6px 0 4px", color: "var(--el-text-primary)" }}
          >
            Analytics
          </h1>
          <p className="text-[13.5px] max-w-xl" style={{ color: "var(--muted-foreground)", margin: 0 }}>
            Aktivitet og tendenser for dit arbejdsområde · seneste 30 dage
          </p>
        </div>
        <AnalyticsClient data={data} />
      </div>
    </div>
  );
}
