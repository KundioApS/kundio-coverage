import type { Json } from "@/types/database";
import type { AiDossierV1 } from "@/lib/ai/dossier/schema";
import type { AiIntakePolicy } from "@/lib/ai/policy/intake-policy";
 
export interface AiBusinessContext {
  id: string;
  name: string;
  description: string | null;
  openingHours: Json;
  websiteUrl: string | null;
  /** Extracted from sitemap crawl; null until crawl has run */
  aiContextSummary: string[] | null;
}
 
export interface AiFaqContext {
  question: string;
  answer: string;
}
 
export interface AiServiceContext {
  name: string;
  description: string | null;
}
 
export interface AiServiceAreaContext {
  name: string;
}
 
export interface AiBusinessHoursContext {
  day_of_week: string;
  hours_text: string;
}
 
export interface AiCallbackWindowContext {
  day_of_week: string;
  start_time: string;
  end_time: string;
  is_active?: boolean;
}
 
export interface AiHistoryMessage {
  direction: "inbound" | "outbound";
  origin: "ai" | "human" | "system";
  body: string;
}
 
export interface AiConversationContext {
  business: AiBusinessContext;
  faqs: AiFaqContext[];
  services: AiServiceContext[];
  serviceAreas: AiServiceAreaContext[];
  businessHours: AiBusinessHoursContext[];
  callbackWindows: AiCallbackWindowContext[];
  history: AiHistoryMessage[];
  /** Persistent lead qualification state from leads.metadata.ai_dossier */
  dossier: AiDossierV1 | null;
  /** Intake policy (generic constant in v1) */
  policy: AiIntakePolicy;
  /** How many AI-origin outbound messages are in this conversation so far */
  aiQuestionsAsked: number;
  /** Current lead status */
  leadStatus: string;
  inboundMessageId: string;
  inboundBody: string;
  inboundProviderKind: string | null;
  inboundProviderRef: string | null;
}
