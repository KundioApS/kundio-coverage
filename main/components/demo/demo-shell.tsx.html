"use client";
 
import { useCallback, useEffect, useRef } from "react";
import { useRouter, useSearchParams } from "next/navigation";
import { AnimatePresence } from "motion/react";
import { TOTAL_DEMO_STEPS } from "@/lib/demo/data";
import { trackDemoEvent } from "@/lib/analytics/demo-tracker";
import { useDemoState } from "./demo-state-provider";
import { DemoHeader } from "./demo-header";
import { DemoBottomNav } from "./demo-bottom-nav";
import { DemoStep } from "./demo-step";
import { DemoLeadForm } from "./demo-lead-form";
import { DemoStepWidget } from "./demo-step-widget";
import { DemoStepInbox } from "./demo-step-inbox";
import { DemoStepPipeline } from "./demo-step-pipeline";
import { DemoStepAnalytics } from "./demo-step-analytics";
import { DemoResults } from "./demo-results";
 
import "./demo-shell.css";
 
export function DemoShell() {
  const searchParams = useSearchParams();
  const router = useRouter();
  const { gatePassed, sessionReady } = useDemoState();
 
  const rawStep = parseInt(searchParams.get("step") ?? "1", 10);
  const step = isNaN(rawStep) ? 1 : Math.min(Math.max(rawStep, 1), TOTAL_DEMO_STEPS);
  const effectiveStep = !gatePassed && step > 1 ? 1 : step;
 
  const stepEnteredAt = useRef<number | null>(null);
 
  useEffect(() => {
    if (!sessionReady) return;
    stepEnteredAt.current = Date.now();
    void trackDemoEvent("demo_step_viewed", { step: effectiveStep });
    if (effectiveStep === TOTAL_DEMO_STEPS) {
      void trackDemoEvent("demo_completed");
    }
  }, [effectiveStep, sessionReady]);
 
  const goToStep = useCallback(
    (next: number) => {
      const dwell = stepEnteredAt.current ? Date.now() - stepEnteredAt.current : 0;
      void trackDemoEvent("demo_step_completed", { step: effectiveStep, dwell_ms: dwell });
      const params = new URLSearchParams(searchParams.toString());
      params.set("step", String(next));
      router.push(`/demo?${params.toString()}`);
    },
    [effectiveStep, router, searchParams],
  );
 
  useEffect(() => {
    if (!sessionReady) return;
    if (!gatePassed && step > 1) {
      const params = new URLSearchParams(searchParams.toString());
      params.set("step", "1");
      router.replace(`/demo?${params.toString()}`);
    }
  }, [gatePassed, router, searchParams, sessionReady, step]);
 
  function renderStep() {
    switch (effectiveStep) {
      case 1:
        return (
          <DemoStep stepKey={1}>
            <DemoLeadForm onDone={() => goToStep(2)} />
          </DemoStep>
        );
      case 2:
        return (
          <DemoStep stepKey={2}>
            <DemoStepWidget onComplete={() => goToStep(3)} />
          </DemoStep>
        );
      case 3:
        return (
          <DemoStep stepKey={3}>
            <DemoStepInbox onComplete={() => goToStep(4)} />
          </DemoStep>
        );
      case 4:
        return (
          <DemoStep stepKey={4}>
            <DemoStepPipeline onComplete={() => goToStep(5)} />
          </DemoStep>
        );
      case 5:
        return (
          <DemoStep stepKey={5}>
            <DemoStepAnalytics onComplete={() => goToStep(6)} />
          </DemoStep>
        );
      case 6:
        return (
          <DemoStep stepKey={6}>
            <DemoResults />
          </DemoStep>
        );
      default:
        return null;
    }
  }
 
  const showChrome = effectiveStep !== 1;
  const isDashboardStep = effectiveStep >= 3 && effectiveStep <= 5;
 
  if (!sessionReady) {
    return <div className="kl-demo min-h-screen bg-[var(--bg)]" />;
  }
 
  return (
    <>
      {showChrome ? <DemoHeader /> : null}
 
      <main
        className={[
          "kl-demo flex min-h-[calc(100vh-56px)] flex-col bg-[var(--bg)] text-[var(--fg)]",
          showChrome ? "pb-[100px]" : "min-h-screen pb-0",
          isDashboardStep ? "kl-demo--dashboard overflow-hidden" : "",
        ]
          .filter(Boolean)
          .join(" ")}
      >
        <div
          className={[
            "kl-demo-stage flex flex-1 flex-col",
            showChrome ? "px-4 py-7 sm:px-7" : "p-0",
          ].join(" ")}
        >
          {showChrome ? (
            <div className="mx-auto flex w-full max-w-[980px] flex-1 flex-col overflow-hidden rounded-[var(--r-lg)] border border-[var(--border)] bg-[var(--bg-raised)] shadow-[var(--shadow-lg)]">
              <AnimatePresence mode="wait">{renderStep()}</AnimatePresence>
            </div>
          ) : (
            <AnimatePresence mode="wait">{renderStep()}</AnimatePresence>
          )}
        </div>
 
        {showChrome ? (
          <DemoBottomNav
            step={effectiveStep}
            gatePassed={gatePassed}
            onBack={() => goToStep(effectiveStep - 1)}
            onNext={() => goToStep(effectiveStep + 1)}
            onJump={(s) => goToStep(s)}
          />
        ) : null}
      </main>
    </>
  );
}
