async function loadOverview(): Promise<OverviewData> {
  try {
    const [templates, recentSends, newsletters, counts] = await Promise.all([
      listTemplates({ templateType: "admin_email", includeArchived: false }),
      listAdminSends({ limit: 12 }),
      listNewsletters({ includeArchived: true }),
      getSubscriptionCounts(),
    ]);
 
    const newsletterSendSets = await Promise.all(
      newsletters.slice(0, 40).map((newsletter) => listNewsletterSends(newsletter.id)),
    );
    const newsletterSends = newsletterSendSets.flat();
    const cutoffMs = Date.now() - 7 * 24 * 60 * 60 * 1000;
    const sent7d = recentSends.filter(
      (send) =>
        send.status === "sent" &&
        send.sentAt &&
        new Date(send.sentAt).getTime() >= cutoffMs,
    ).length;
 
    return {
      templates,
      recentSends,
      newsletters,
      newsletterSends,
      sent7d,
      counts,
      error: null,
    };
  } catch (err) {
    return {
      templates: [],
      recentSends: [],
      newsletters: [],
      newsletterSends: [],
      sent7d: 0,
      counts: { subscribed: 0, unsubscribed: 0 },
      error: err instanceof Error ? err.message : "Ukendt fejl",
    };
  }
}

function buildActivity(data: OverviewData): ActivityItem[] {
  const newsletterTitle = new Map(
    data.newsletters.map((newsletter) => [newsletter.id, newsletter.title]),
  );
 
  const oneOff = data.recentSends.map<ActivityItem>((send) => ({
    id: `admin-send-${send.id}`,
    title: send.isTest ? "Test-email sendt" : "Kundeemail sendt",
    detail: send.subject,
    meta: `${send.recipientEmail} · ${formatDateTime(send.sentAt ?? send.createdAt)}`,
    href: "/admin/emails/history",
    status:
      send.status === "sent"
        ? "sent"
        : send.status === "failed"
          ? "failed"
          : "pending",
    at: send.sentAt ?? send.createdAt,
  }));
 
  const newsletterSends = data.newsletterSends.map<ActivityItem>((send) => ({
    id: `newsletter-send-${send.id}`,
    title: "Nyhedsbrev-afsendelse",
    detail: newsletterTitle.get(send.newsletter_id) ?? "Nyhedsbrev",
    meta: `${send.sent_count}/${send.recipient_count} sendt · ${send.failed_count} fejlede`,
    href: `/admin/emails/newsletters/${send.newsletter_id}`,
    status:
      send.status === "completed"
        ? "sent"
        : send.status === "failed" || send.status === "partially_failed"
          ? "failed"
          : "pending",
    at: send.completed_at ?? send.started_at ?? send.created_at,
  }));
 
  const newsletters = data.newsletters.slice(0, 8).map<ActivityItem>((newsletter) => ({
    id: `newsletter-${newsletter.id}`,
    title:
      newsletter.status === "sent"
        ? "Nyhedsbrev sendt"
        : newsletter.status === "archived"
          ? "Nyhedsbrev arkiveret"
          : "Nyhedsbrev opdateret",
    detail: newsletter.title,
    meta: formatDateTime(newsletter.sentAt ?? newsletter.updatedAt),
    href: `/admin/emails/newsletters/${newsletter.id}`,
    status: newsletter.status,
    at: newsletter.sentAt ?? newsletter.updatedAt,
  }));
 
  return [...oneOff, ...newsletterSends, ...newsletters]
    .sort((a, b) => new Date(b.at).getTime() - new Date(a.at).getTime())
    .slice(0, 7);
}
