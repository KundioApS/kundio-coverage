type TurnRow = {
  id: string;
  created_at: string;
  conversation_id: string;
  inbound_message_id: string;
  decision: unknown;
  model: string | null;
};
 
function asRecord(value: unknown): Record<string, unknown> {
  return value && typeof value === "object" && !Array.isArray(value)
    ? (value as Record<string, unknown>)
    : {};
}
 
function readTraceSteps(decision: unknown): TraceStep[] {
  const audit = asRecord(asRecord(decision).audit);
  const trace = asRecord(audit.trace);
  const steps = trace.steps;
  return Array.isArray(steps) ? (steps as TraceStep[]) : [];
}
 
function readOutcome(decision: unknown): string {
  const record = asRecord(decision);
  const type = typeof record.type === "string" ? record.type : "unknown";
  if (type === "reply" && typeof record.body === "string") return record.body;
  if (type === "handoff" && typeof record.reason === "string") {
    return `handoff: ${formatHandoffReason(record.reason) ?? record.reason}`;
  }
  if (type === "silent" && typeof record.reason === "string") return `silent: ${record.reason}`;
  return type;
}
 
function fmt(value: string): string {
  return new Intl.DateTimeFormat("da-DK", {
    dateStyle: "medium",
    timeStyle: "short",
  }).format(new Date(value));
}
 
function joinIds(ids?: string[]): string {
  return ids && ids.length > 0 ? ids.join(", ") : "-";
}
 
function yesNo(value: boolean | undefined): string {
  return value ? "ja" : "nej";
}
