function SettingRow({
  controlId,
  label,
  hint,
  children,
}: {
  controlId?: string;
  label: string;
  hint?: string;
  children: ReactNode;
}) {
  const labelId = controlId ? undefined : `admin-setting-${label.replace(/\s+/g, "-").toLowerCase()}-label`;
  const hintId = hint ? `admin-setting-${label.replace(/\s+/g, "-").toLowerCase()}-hint` : undefined;
 
  return (
    <div
      data-setting-row={label}
      className="grid gap-3 py-4 sm:grid-cols-[minmax(0,1fr)_minmax(12rem,auto)] sm:items-center"
      style={{
        borderBottomColor: "var(--line)",
        borderBottomStyle: "solid",
        borderBottomWidth: 1,
      }}
    >
      <div>
        {controlId ? (
          <label
            htmlFor={controlId}
            style={{ fontSize: 13.5, fontWeight: 600, display: "block" }}
          >
            {label}
          </label>
        ) : (
          <b
            id={labelId}
            style={{ fontSize: 13.5, fontWeight: 600, display: "block" }}
          >
            {label}
          </b>
        )}
        {hint && (
          <small
            id={hintId}
            style={{
              fontSize: 12.5,
              color: "var(--ink-mute)",
              display: "block",
              marginTop: 2,
              lineHeight: 1.4,
            }}
          >
            {hint}
          </small>
        )}
      </div>
      <div className="min-w-0 justify-self-start sm:justify-self-end">{children}</div>
    </div>
  );
}

                <div className="notice notice-warning" style={{ margin: "12px 0" }}>

              )}
              <SettingRow label="Påkræv to-faktor" hint="Kommer senere. Kontrollen er deaktiveret indtil login-flowet håndhæver MFA.">

                  onChange={(v) => set("adminRequireMfa", v)}

                <Toggle label="Bekræft destruktive handlinger" on={settings.confirmDestructiveActions} onChange={(v) => set("confirmDestructiveActions", v)} />
              </SettingRow>
            </div>
          )}
 
          {section === "notify" && (
            <div className="card-bd">
              <h3 style={{ margin: "0 0 4px", fontSize: 16, fontWeight: 700 }}>Notifikationer</h3>

              <SettingRow label="Fejlet betaling" hint="Send alert når en Stripe-betaling fejler.">
                <Toggle label="Fejlet betaling" on={settings.notifyFailedPayment} onChange={(v) => set("notifyFailedPayment", v)} />
              </SettingRow>
              <SettingRow label="Twilio-fejl" hint="Send alert når provisionering fejler.">
                <Toggle label="Twilio-fejl" on={settings.notifyTwilioFailure} onChange={(v) => set("notifyTwilioFailure", v)} />
              </SettingRow>
              <SettingRow label="Ny tilmelding" hint="Send alert når en virksomhed bliver oprettet.">
                <Toggle label="Ny tilmelding" on={settings.notifyNewSignup} onChange={(v) => set("notifyNewSignup", v)} />
              </SettingRow>
              <SettingRow controlId="admin-slack-webhook" label="Slack webhook" hint="Send alerts til Slack #admin-alerts.">
                <div className="flex w-full flex-col gap-2 sm:flex-row sm:items-center">
                  <input id="admin-slack-webhook" className="input mono" placeholder={settings.slackWebhookConfigured ? "Konfigureret · skriv ny URL for at erstatte" : "https://hooks.slack.com/services/..."} value={settings.slackWebhookUrl} onChange={(e) => set("slackWebhookUrl", e.target.value)} style={{ ...controlStyle(320), fontSize: 12 }} />
                  <button type="button" className="btn btn-outline btn-sm" onClick={testSlack} disabled={testingSlack || !savedSettings.slackWebhookConfigured}>
                    <I.Bell /> {testingSlack ? "Sender…" : "Send test"}
                  </button>
                </div>
              </SettingRow>
            </div>
          )}
 
          <div
            className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-end"
            style={{
              padding: "14px 20px",
              borderTop: "1px solid var(--line)",
              background: "var(--surface-warm)",
            }}
          >
            <span
              role="status"
              aria-live="polite"
              className="min-h-4 text-xs font-medium sm:mr-auto"
              style={{
                color: saving ? "var(--ink-mute)" : "var(--ok-ink)",
              }}
            >
              {saving ? "Gemmer ændringer…" : ""}
            </span>
            <button type="button" className="btn btn-outline btn-sm" onClick={() => { setSettings(savedSettings); setNotice(null); }} disabled={saving}>Annuller</button>
            <button type="button" className="btn btn-primary btn-sm" onClick={save} disabled={saving}><I.Check /> {saving ? "Gemmer…" : "Gem ændringer"}</button>
          </div>
        </div>
      </div>
    </div>
  );
