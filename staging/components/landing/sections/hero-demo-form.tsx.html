"use client";
 
import { useState, useTransition } from "react";
import { useRouter } from "next/navigation";
 
import { KundioButton } from "@/components/landing/button";
import { KundioInput } from "@/components/landing/public-ui";
import { storePassedDemoGate } from "@/lib/demo/session";
import { newEventId, trackLead } from "@/lib/meta/track-client";
import { KlArrow, KlTickLg } from "./icons";
 
const EMAIL_PATTERN = /\S+@\S+\.\S+/;
 
export function HeroDemoForm() {
  const router = useRouter();
  const [email, setEmail] = useState("");
  const [submitting, setSubmitting] = useState(false);
  const [isPending, startTransition] = useTransition();
 
  const normalizedEmail = email.trim();
  const isValid = EMAIL_PATTERN.test(normalizedEmail);
  const isBusy = submitting || isPending;
 
  function handleSubmit(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();
    if (!isValid || isBusy) {
      event.currentTarget.reportValidity();
      return;
    }
 
    setSubmitting(true);
    storePassedDemoGate({ email: normalizedEmail });
 
    // TODO(consent): only fire pixel/CAPI tracking when marketing consent is granted.
    const eventId = newEventId();
    void fetch("/api/marketing/demo-lead", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ email: normalizedEmail, eventId }),
    })
      .then(() => {
        trackLead(eventId, { content_name: "Interactive Demo" });
      })
      .catch((err) => {
        console.warn("[hero] demo lead submission failed", err);
      });
 
    startTransition(() => {
      router.push("/demo?step=2");
    });
  }
 
  return (
    <>
      <form
        className="mx-auto grid w-full max-w-[640px] grid-cols-1 gap-2.5 min-[560px]:grid-cols-[minmax(0,1fr)_auto]"
        onSubmit={handleSubmit}
      >
        <label htmlFor="hero-demo-email" className="sr-only">
          E-mail
        </label>
        <KundioInput
          id="hero-demo-email"
          type="email"
          value={email}
          onChange={(event) => setEmail(event.target.value)}
          placeholder="Din e-mail"
          required
          autoComplete="email"
          inputMode="email"
          aria-label="E-mail til demo"
          className="h-[52px] border-[1.5px] border-[var(--border-strong)] bg-[var(--ink-50)] px-4 text-[16px]"
        />
        <KundioButton
          type="submit"
          size="lg"
          disabled={isBusy}
          className="h-[52px] w-full px-6 disabled:opacity-60 min-[560px]:w-auto min-[560px]:min-w-[150px]"
        >
          {isBusy ? "Starter…" : "Prøv demoen"}
          <KlArrow />
        </KundioButton>
      </form>
      <ul className="mx-auto mt-3 mb-10 flex max-w-[640px] list-none flex-wrap items-center justify-center gap-x-4 gap-y-1.5 text-[13px] leading-[1.4] text-brand-ink/70 min-[560px]:justify-start">
        {["Ingen oprettelse", "2 min", "Ingen kortoplysninger"].map((label) => (
          <li key={label} className="inline-flex items-center gap-1.5 whitespace-nowrap">
            <span aria-hidden="true" className="inline-flex shrink-0 text-brand-ink">
              <KlTickLg />
            </span>
            {label}
          </li>
        ))}
      </ul>
    </>
  );
}
