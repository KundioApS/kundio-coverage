function ConversationRow({
  item,
  isFirst,
  index,
  animate,
}: {
  item: ConversationItem;
  isFirst: boolean;
  index: number;
  animate: boolean;
}) {
  return (
    <motion.li
      className="kl-dash-row"
      initial={animate ? { opacity: 0, y: 4 } : false}
      animate={{ opacity: 1, y: 0 }}
      transition={
        animate
          ? { duration: 0.22, delay: index * 0.04, ease: [0.22, 1, 0.36, 1] }
          : { duration: 0 }
      }
    >
      <span className={`kl-dash-avatar${isFirst ? " is-accent" : ""}`} aria-hidden>
        <Image
          src={item.avatarSrc}
          alt=""
          width={32}
          height={32}
          sizes="32px"
          className="kl-dash-avatar-image"
        />
      </span>
      <span className="kl-dash-row-main">
        <span className="kl-dash-row-line">
          <strong className="kl-dash-row-name">{item.name}</strong>
          <span className="kl-dash-row-time">{item.time}</span>
          <SourceIcon kind={item.source} />
        </span>
        <span className="kl-dash-row-preview">{item.preview}</span>
        <span
          className={`kl-dash-pill kl-dash-pill--${
            item.status === "Ny lead" ? "lead-new" : "neutral"

      </span>
    </motion.li>
  );
}
 
function ReviewRow({
  item,
  isFirst,
  index,
  animate,
}: {
  item: ReviewItem;
  isFirst: boolean;
  index: number;
  animate: boolean;
}) {
  const [reply, setReply] = useState("");
  const inputId = `kl-dash-review-reply-${item.id}`;
 
  function handleSubmit(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setReply("");
  }
 
  return (
    <motion.li
      className="kl-dash-row kl-dash-row--review"
      initial={animate ? { opacity: 0, y: 4 } : false}
      animate={{ opacity: 1, y: 0 }}
      transition={
        animate
          ? { duration: 0.22, delay: index * 0.04, ease: [0.22, 1, 0.36, 1] }
          : { duration: 0 }
      }
    >
      <span className={`kl-dash-avatar${isFirst ? " is-accent" : ""}`} aria-hidden>
        <Image
          src={item.avatarSrc}
          alt=""
          width={32}
          height={32}
          sizes="32px"
          className="kl-dash-avatar-image"
        />
      </span>
      <span className="kl-dash-row-main">
        <span className="kl-dash-row-line">
          <strong className="kl-dash-row-name">{item.name}</strong>
          <span className="kl-dash-row-time">{item.time}</span>
          <SourceIcon kind={item.platform} />
        </span>
        <span className="kl-dash-stars" aria-label={`${item.rating} af 5 stjerner`}>
          {Array.from({ length: 5 }).map((_, i) => (
            <Star
              key={i}
              size={11}
              className={i < item.rating ? "is-on" : ""}
              fill={i < item.rating ? "currentColor" : "none"}
              aria-hidden
            />
          ))}
        </span>
        <span className="kl-dash-row-preview">{item.preview}</span>
        {item.reply ? (
          <span className="kl-dash-reply-shown">
            <span className="kl-dash-reply-shown-label">
              Du svarede · {item.reply.time}
            </span>
            <span className="kl-dash-reply-shown-text">{item.reply.text}</span>
          </span>
        ) : null}
        <form className="kl-dash-reply" onSubmit={handleSubmit}>
          <label className="kl-dash-search-label" htmlFor={inputId}>
            Svar til {item.name}
          </label>
          <input
            id={inputId}
            type="text"
            className="kl-dash-reply-input"
            placeholder={
              item.reply ? "Skriv et opfølgende svar…" : `Svar til ${item.name.split(" ")[0]}…`
            }
            value={reply}
            onChange={(event) => setReply(event.target.value)}
          />
          <button
            type="submit"
            className="kl-dash-reply-send"
            aria-label="Send svar"
            disabled={reply.trim().length === 0}
          >
            <Send size={12} aria-hidden />
          </button>
        </form>
      </span>
    </motion.li>
  );
}
 
const CYCLE_MS = 6000;
const PANEL_EASE = [0.22, 1, 0.36, 1] as const;
 
export function DashboardVisualTabs() {
  const reduceMotion = useReducedMotion() ?? false;
  const [activeInbox, setActiveInbox] = useState<"samtaler" | "anmeldelser">("samtaler");
  const [isHovered, setIsHovered] = useState(false);
  const [manualOverride, setManualOverride] = useState(false);
  const sectionRef = useRef<HTMLDivElement | null>(null);
  const isVisible = useInView(sectionRef, { amount: 0.4 });
 
  const isPaused = isHovered || !isVisible || reduceMotion || manualOverride;
 
  useEffect(() => {
    if (isPaused) return;
    const id = setTimeout(() => {
      setActiveInbox((prev) => (prev === "samtaler" ? "anmeldelser" : "samtaler"));
    }, CYCLE_MS);
    return () => clearTimeout(id);
  }, [activeInbox, isPaused]);
 
  const handleManual = (next: "samtaler" | "anmeldelser") => {
    setManualOverride(true);
    setActiveInbox(next);
  };
 
  const animateRows = !reduceMotion;
 
  return (
    <div
      ref={sectionRef}
      className="kl-dashboard-visual kl-dashboard-visual--compact kl-dash-v2"
      aria-label="Kundio dashboard"
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}
      onFocus={() => setIsHovered(true)}
      onBlur={() => setIsHovered(false)}
    >
      <div className="kl-dash-frame kl-dashboard-visual-frame kl-dash-v2-frame">
        <div className="kl-dash-panel-shell kl-dash-shell--split">
          <aside className="kl-dash-sidebar" aria-label="Indbakker">
            <div className="kl-dash-sidebar-brand">
              <BrandLogo height={20} alt="Kundio" />
            </div>
            <nav className="kl-dash-sidebar-nav" aria-label="Skift indbakke">
              <button
                type="button"
                className={
                  activeInbox === "samtaler"
                    ? "kl-dash-sidebar-item is-active"
                    : "kl-dash-sidebar-item"
                }
                aria-current={activeInbox === "samtaler" ? "page" : undefined}
                onClick={() => handleManual("samtaler")}
              >
                <MessageCircle size={14} aria-hidden />
                <span className="kl-dash-sidebar-label">Samtaler</span>
              </button>
              <button
                type="button"
                className={
                  activeInbox === "anmeldelser"
                    ? "kl-dash-sidebar-item is-active"
                    : "kl-dash-sidebar-item"
                }
                aria-current={activeInbox === "anmeldelser" ? "page" : undefined}
                onClick={() => handleManual("anmeldelser")}
              >
                <Star size={14} aria-hidden />
                <span className="kl-dash-sidebar-label">Anmeldelser</span>
              </button>
            </nav>
          </aside>
 
          <AnimatePresence mode="wait" initial={false}>
            {activeInbox === "samtaler" ? (
              <motion.section
                key="samtaler"
                className="kl-dash-panel kl-dash-panel--inbox"
                aria-label="Samtaler indbakke"
                initial={reduceMotion ? false : { opacity: 0, y: 6 }}
                animate={{ opacity: 1, y: 0 }}
                exit={reduceMotion ? { opacity: 1 } : { opacity: 0, y: -6 }}
                transition={{ duration: reduceMotion ? 0 : 0.28, ease: PANEL_EASE }}
              >
                <header className="kl-dash-toolbar">
                  <div className="kl-dash-toolbar-actions">
                    <label className="kl-dash-search">
                      <Search size={12} aria-hidden />
                      <input
                        type="text"
                        readOnly
                        placeholder="Søg"
                        aria-label="Søg i samtaler"
                        tabIndex={-1}
                      />
                    </label>
                    <button
                      type="button"
                      className="kl-dash-action kl-dash-action--icon"
                      aria-label="Ny besked"
                      tabIndex={-1}
                    >
                      <SquarePen size={14} aria-hidden />
                    </button>
                  </div>
                </header>
 
                <div className="kl-dash-tabs" role="tablist" aria-label="Samtale-status">
                  <span className="kl-dash-tab is-active" role="tab" aria-selected="true">
                    Åbne <em>5</em>
                  </span>
                  <span className="kl-dash-tab" role="tab" aria-selected="false">
                    Lukkede <em>12</em>
                  </span>
                </div>
 
                <ul className="kl-dash-list" role="list">
                  {SAMTALER_ITEMS.map((item, index) => (
                    <ConversationRow
                      key={item.id}
                      item={item}
                      isFirst={index === 0}
                      index={index}
                      animate={animateRows}
                    />
                  ))}
                </ul>
              </motion.section>
            ) : (
              <motion.section
                key="anmeldelser"
                className="kl-dash-panel kl-dash-panel--inbox"
                aria-label="Anmeldelser indbakke"
                initial={reduceMotion ? false : { opacity: 0, y: 6 }}
                animate={{ opacity: 1, y: 0 }}
                exit={reduceMotion ? { opacity: 1 } : { opacity: 0, y: -6 }}
                transition={{ duration: reduceMotion ? 0 : 0.28, ease: PANEL_EASE }}
              >
                <header className="kl-dash-toolbar">
                  <div className="kl-dash-toolbar-actions">
                    <label className="kl-dash-search">
                      <Search size={12} aria-hidden />
                      <input
                        type="text"
                        readOnly
                        placeholder="Søg"
                        aria-label="Søg i anmeldelser"
                        tabIndex={-1}
                      />
                    </label>
                    <button
                      type="button"
                      className="kl-dash-action kl-dash-action--icon"
                      aria-label="Filtrer anmeldelser"
                      tabIndex={-1}
                    >
                      <SlidersHorizontal size={14} aria-hidden />
                    </button>
                  </div>
                </header>
 
                <div className="kl-dash-tabs" role="tablist" aria-label="Anmeldelse-filter">
                  <span className="kl-dash-tab is-active" role="tab" aria-selected="true">
                    Alle <em>5</em>
                  </span>
                  <span className="kl-dash-tab" role="tab" aria-selected="false">
                    Google <em>3</em>
                  </span>
                  <span className="kl-dash-tab" role="tab" aria-selected="false">
                    Trustpilot <em>1</em>
                  </span>
                </div>
 
                <ul className="kl-dash-list" role="list">
                  {ANMELDELSER_ITEMS.map((item, index) => (
                    <ReviewRow
                      key={item.id}
                      item={item}
                      isFirst={index === 0}
                      index={index}
                      animate={animateRows}
                    />
                  ))}
                </ul>
              </motion.section>
            )}
          </AnimatePresence>
        </div>
      </div>
    </div>
  );
}
