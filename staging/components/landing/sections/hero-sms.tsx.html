  const elapsed = Math.max(0, elapsedMs);

    const text = WIDGET_FIELDS[fieldIdx];

      return {
        field: fieldIdx as 0 | 1 | 2,
        chars: Math.max(0, Math.floor((elapsed - cursor) / WIDGET_TYPE_INTERVAL_MS)),
        done: false,
      };
