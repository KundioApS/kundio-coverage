"use client";

}

}
 
function DropdownMenuGroup({
  ...props
}: React.ComponentProps<typeof DropdownMenuPrimitive.Group>) {

}

  ...props

  return <DropdownMenuPrimitive.Portal data-slot="dropdown-menu-portal" {...props} />;
}

  align = "end",
  sideOffset = 8,
  ...props

      />

}

  ...props
}: React.ComponentProps<typeof DropdownMenuPrimitive.Item>) {
  return (
    <DropdownMenuPrimitive.Item
      data-slot="dropdown-menu-item"
      className={cn(
        "relative flex min-h-10 cursor-pointer select-none items-center gap-2 rounded-[10px] px-3 py-2 text-[13px] outline-none transition-[background-color,color,opacity] duration-150 ease-out hover:bg-[color-mix(in_srgb,var(--el-brand-tint)_56%,var(--el-surface)_44%)] focus:bg-[color-mix(in_srgb,var(--el-brand-tint)_56%,var(--el-surface)_44%)] focus:text-[var(--el-ink)] motion-reduce:transition-none data-[highlighted]:bg-[color-mix(in_srgb,var(--el-brand-tint)_56%,var(--el-surface)_44%)] data-[highlighted]:text-[var(--el-ink)] data-[disabled]:pointer-events-none data-[disabled]:cursor-not-allowed data-[disabled]:opacity-45",
        className,

      {...props}
    />

}
 
function DropdownMenuLabel({
