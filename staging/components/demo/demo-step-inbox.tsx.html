function LeadRow({
  name,
  preview,
  time,
  badge,
  selected,
  faded,
  tourId,
  onClick,
}: {
  name: string;
  preview: string;
  time: string;
  badge: string;
  selected?: boolean;
  faded?: boolean;
  tourId?: string;
  onClick?: () => void;
}) {
  const badgeColor =
    badge === "Ny"
      ? { bg: "var(--el-brand-tint, #f5ede8)", text: "var(--el-brand-ink, oklch(0.38 0.10 40))" }
      : badge === "Kvalificeret"
      ? { bg: "#e4eee4", text: "#2f5a40" }
      : { bg: "var(--el-surface-warm-2, #e9e5dd)", text: "var(--el-ink-soft)" };
 
  return (
    <div
      data-tour-id={tourId}
      onClick={onClick}
      style={{
        padding: "10px 18px",
        borderBottom: "1px solid var(--el-line)",
        cursor: onClick ? "pointer" : "default",
        background: selected ? "var(--el-surface-warm, #f1eee8)" : "transparent",
        opacity: faded ? 0.45 : 1,
        borderLeft: selected ? "3px solid var(--el-brand)" : "3px solid transparent",
        transition: "background 0.14s",
        userSelect: "none",
      }}
    >
      <div style={{ display: "flex", alignItems: "flex-start", justifyContent: "space-between", gap: 6, marginBottom: 4 }}>
        <span style={{ fontSize: 13.5, fontWeight: 600, color: "var(--el-ink)", overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap" }}>
          {name}
        </span>
        <span style={{ fontSize: 11, fontFamily: "var(--font-mono)", color: "var(--el-ink-mute)", flexShrink: 0 }}>
          {time} siden
        </span>
      </div>

}
 
function MessageBubble({ role, text, time }: { role: "them" | "ai"; text: string; time: string }) {
  const isAi = role === "ai";
  return (

      {!isAi && (
        <span style={{ fontSize: 10, textTransform: "uppercase", letterSpacing: "0.05em", color: "var(--el-ink-faint)", fontFamily: "var(--font-mono)" }}>
          Kunde
        </span>
      )}
      <div
        style={{
          padding: "10px 14px",
          borderRadius: 18,
          fontSize: 13,
          lineHeight: 1.55,
          ...(isAi
            ? { background: "var(--el-brand)", color: "#fff", borderBottomRightRadius: 4 }
            : { background: "#fff", border: "1px solid var(--el-line)", color: "var(--el-ink)", borderBottomLeftRadius: 4 }),
        }}
      >
        {text}
      </div>
      <span style={{ fontSize: 10, color: "var(--el-ink-faint)", fontFamily: "var(--font-mono)" }}>
        {isAi ? "Hansen VVS AI · " : ""}{time}
      </span>
    </div>
  );
}

export function DemoStepInbox({ onComplete }: Props) {
  const [detailOpen, setDetailOpen] = useState(false);
  const [handed, setHanded] = useState(false);
 
  const scenes = makeScenes();
  const { scene, sceneIdx, total, advance } = useTour(scenes, onComplete);
 
  const messages = buildDemoScript(DEMO_LEAD.name, DEMO_LEAD.service);
 
  function handleLeadClick() {
    if (scene?.waitForClick && scene.targetId === "inbox-lead-row") {
      setDetailOpen(true);
      advance();
    }
  }
 
  function handleHandoffClick() {
    if (scene?.waitForClick && scene.targetId === "inbox-handoff-btn") {
      setHanded(true);
      advance();
    }
  }
 
  return (
    <div style={{ flex: 1, display: "flex", flexDirection: "column", minHeight: 0 }}>
      <DemoDashboardShell activePath="/inbox">
        <div style={{ display: "grid", gridTemplateColumns: detailOpen ? "200px 280px 1fr" : "200px 1fr", height: "100%", minHeight: 0 }}>
 
          {/* Left: view filters */}
          <div style={{ background: "var(--el-surface)", borderRight: "1px solid var(--el-line)", padding: "16px 0", overflowY: "auto" }}>
            <p style={{ padding: "0 14px 8px", fontSize: 10, fontWeight: 600, textTransform: "uppercase", letterSpacing: "0.12em", color: "var(--el-ink-faint)", fontFamily: "var(--font-mono)" }}>
              Views
            </p>
            {[
              { id: "nye", label: "Nye", count: 1 },
              { id: "kvalificerede", label: "Kvalificerede", count: 2 },
              { id: "klar", label: "Klar til opkald", count: 1 },
              { id: "alle", label: "Alle", count: 4 },
            ].map((v) => (
              <div
                key={v.id}
                style={{
                  padding: "7px 14px",
                  fontSize: 13,
                  display: "flex",
                  justifyContent: "space-between",
                  alignItems: "center",
                  borderRadius: 7,
                  margin: "0 6px 1px",
                  background: v.id === "nye" ? "var(--el-surface-warm)" : "transparent",
                  color: v.id === "nye" ? "var(--el-ink)" : "var(--el-ink-mute)",
                  cursor: "default",
                }}
              >
                <span>{v.label}</span>
                <span style={{ fontSize: 11, fontFamily: "var(--font-mono)", opacity: 0.6 }}>{v.count}</span>
              </div>
            ))}
          </div>
 
          {/* Center: conversation list */}
          <div style={{ borderRight: "1px solid var(--el-line)", display: "flex", flexDirection: "column", minHeight: 0, background: "var(--el-surface)" }}>
            <div style={{ padding: "14px 18px 10px", borderBottom: "1px solid var(--el-line)", display: "flex", alignItems: "baseline", gap: 8, flexShrink: 0 }}>
              <span style={{ fontSize: 14, fontWeight: 600, color: "var(--el-ink)" }}>Indbakke</span>
              <span style={{ fontSize: 12, color: "var(--el-ink-faint)", fontFamily: "var(--font-mono)" }}>3 kundeemner</span>
            </div>
 
            <div style={{ overflowY: "auto", flex: 1 }}>
              <LeadRow
                name={DEMO_LEAD.name}
                preview="Badeværelsesrenovering — ca. 6 m²"
                time="3 min"
                badge="Ny"
                selected={detailOpen}
                tourId="inbox-lead-row"
                onClick={handleLeadClick}
              />
              {FILLER_LEADS.map((l) => (
                <LeadRow key={l.id} name={l.name} preview={l.preview} time={l.time} badge={l.badge} faded />
              ))}
            </div>
          </div>
 
          {/* Right: conversation detail */}
          {detailOpen && (
            <div style={{ display: "flex", minHeight: 0, overflow: "hidden" }}>
              {/* Thread area */}
              <div style={{ flex: 1, display: "flex", flexDirection: "column", minHeight: 0, minWidth: 0 }}>
                {/* Handoff banner */}
                {handed && (
                  <div
                    data-tour-id="inbox-handoff-banner"
                    style={{
                      padding: "9px 16px",
                      background: "var(--el-ink)",
                      color: "#fff",
                      fontSize: 12.5,
                      fontWeight: 500,
                      display: "flex",
                      alignItems: "center",
                      gap: 8,
                      flexShrink: 0,
                    }}
                  >
                    <span style={{ width: 8, height: 8, borderRadius: "50%", background: "#4ade80", flexShrink: 0 }} />
                    Menneskelig kontrol aktiveret — du har overtaget samtalen
                  </div>
                )}
 
                {/* Conversation header */}
                <div style={{ padding: "12px 16px", borderBottom: "1px solid var(--el-line)", display: "flex", alignItems: "center", gap: 10, flexShrink: 0 }}>
                  <div style={{ width: 34, height: 34, borderRadius: "50%", background: "var(--el-surface-warm-2)", display: "grid", placeItems: "center", fontSize: 12, fontWeight: 700, color: "var(--el-ink-soft)", flexShrink: 0 }}>

                    <button
                      type="button"
                      data-tour-id="inbox-handoff-btn"
                      onClick={handleHandoffClick}
                      style={{
                        padding: "6px 14px",
                        borderRadius: 8,
                        background: "var(--el-ink)",
                        color: "#fff",
                        fontSize: 12.5,
                        fontWeight: 600,
                        border: "none",
                        cursor: "pointer",
                        flexShrink: 0,
                      }}
                    >
                      Overtag
                    </button>
                  )}
                </div>
 
                {/* Thread */}
                <div
                  data-tour-id="inbox-thread"
                  style={{
                    flex: 1,
                    overflowY: "auto",
                    padding: "16px",
                    display: "flex",
                    flexDirection: "column",
                    gap: 10,
                    background: "var(--el-surface-raised)",
                  }}
                >
                  {messages.map((msg, i) => (
                    <MessageBubble key={i} role={msg.role} text={msg.text} time={msg.t} />
                  ))}
                </div>
 
                {/* Composer */}
                <div style={{ padding: "10px 14px", borderTop: "1px solid var(--el-line)", background: "var(--el-surface)", flexShrink: 0 }}>
                  <div style={{ borderRadius: 9, padding: "9px 13px", fontSize: 13, color: "var(--el-ink-faint)", background: "var(--el-surface-raised)", border: "1px solid var(--el-line)", fontStyle: "italic" }}>
                    Svar kunden…
                  </div>
                </div>
              </div>
 
              {/* AI Brief sidebar */}
              <AiBriefPanel handed={handed} />
            </div>
          )}
        </div>
      </DemoDashboardShell>
 
      {/* Tour overlay */}
      {scene && (
        <TourOverlay
          scene={scene}
          sceneIdx={sceneIdx}
          total={total}
          onAdvance={advance}
        />
      )}
    </div>
  );
}
