import { createHash, randomBytes, timingSafeEqual } from "node:crypto";
 
/**
 * Newsletter unsubscribe tokens.
 *
 * Generation: 24 random bytes → base64url (32 chars, ~192 bits).
 * Storage: SHA-256 hex of the raw token. The raw token is never persisted —
 * the dispatcher embeds it into the outgoing email URL and forgets it.
 * Verification: timing-safe equality on the hashed input.
 *
 * On unsubscribe the hash is rotated so the link cannot be reused or
 * re-flipped by someone who replayed the URL.
 *
 * Never log the raw token.
 */
export function createUnsubscribeToken(): string {
  return randomBytes(24).toString("base64url");
}
 
export function hashUnsubscribeToken(token: string): string {
  return createHash("sha256").update(token).digest("hex");
}
